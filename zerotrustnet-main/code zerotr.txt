# -------------------------------
# 📊 DETECTION
# -------------------------------
elif menu == "📊 Detection":

    st.title("📊 Detection Dashboard")

    data = pd.read_csv("data.csv")

    model = IsolationForest(contamination=0.3)
    data['anomaly'] = model.fit_predict(data[['login_time', 'file_access']])
    data['Result'] = data['anomaly'].apply(
        lambda x: "Suspicious" if x == -1 else "Normal"
    )

    suspicious_users = data[data['Result'] == "Suspicious"]

    col1, col2 = st.columns(2)
    col1.metric("👥 Total Users", len(data))
    col2.metric("⚠ Suspicious", len(suspicious_users))

    st.subheader("🚨 Threat Monitoring")

    if len(suspicious_users) > 0:
        st.error(f"🚨 {len(suspicious_users)} Suspicious Users Detected!")
    else:
        st.success("✅ All Users Normal")

    # Highlight suspicious
    def highlight(row):
        if row["Result"] == "Suspicious":
            return ['background-color:red; color:white']*len(row)
        return ['']*len(row)

    st.subheader("📂 User Data")
    st.dataframe(data.style.apply(highlight, axis=1))

# -------------------------------
# 📈 ANALYTICS
# -------------------------------
elif menu == "📈 Analytics":

    st.title("📈 Data Analytics")

    data = pd.read_csv("data.csv")

    model = IsolationForest(contamination=0.3)
    data['anomaly'] = model.fit_predict(data[['login_time', 'file_access']])
    data['Result'] = data['anomaly'].apply(
        lambda x: "Suspicious" if x == -1 else "Normal"
    )

    st.subheader("📊 Login Time")
    st.bar_chart(data['login_time'])

    st.subheader("📊 Normal vs Suspicious")
    st.bar_chart(data['Result'].value_counts())

# -------------------------------
# 🔍 CHECK USER
# -------------------------------
elif menu == "🔍 Check User":

    st.title("🔍 Check User Activity")

    login_time = st.number_input("Login Time (0-23)", 0, 23)
    file_access = st.number_input("File Access Count", 0, 100)

    if st.button("Check"):

        data = pd.read_csv("data.csv")

        model = IsolationForest(contamination=0.3)
        model.fit(data[['login_time', 'file_access']])

        result = model.predict([[login_time, file_access]])

        if result[0] == -1:
            st.error("⚠ Suspicious Activity!")
        else:
            st.success("✅ Normal Activity")

    st.info("Suspicious if unusual login time or high file access")
